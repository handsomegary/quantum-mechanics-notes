\notepage{84}{Radial Fourier integral and the contour prescription}
Choose the polar axis in momentum space parallel to $\mathbf x$, and
write $r=|\mathbf x|>0$. Inserting the Fourier-space kernel,
\begin{align*}
 G^{(0)}(r)
 &=\frac1{8\pi^3}\int_0^\infty q^2\dd q\int_0^{2\pi}\dd\phi
    \int_{-1}^{1}\dd(\cos\theta)\,
    \frac{\ee^{\ii qr\cos\theta}}{k^2+\ii\epsilon-q^2}\\
 &=\frac1{4\pi^2}\int_0^\infty
    \frac{\ee^{\ii qr}-\ee^{-\ii qr}}{\ii qr}
    \frac{q^2\dd q}{k^2+\ii\epsilon-q^2}\\
 &=\frac1{4\pi^2}\int_{-\infty}^{\infty}
    \frac{\ee^{\ii qr}}{\ii qr}
    \frac{q^2\dd q}{k^2+\ii\epsilon-q^2}\\
 &=-\frac{\ii}{4\pi^2r}\int_{-\infty}^{\infty}
    \frac{q\,\ee^{\ii qr}}{k^2+\ii\epsilon-q^2}\dd q.
\end{align*}
\editorial{The handwritten last prefactor contains $r^2$ in its
denominator. The preceding line gives a single factor $r$, as shown
above.}
Consider the integrand
\[
 \frac{q\,\ee^{\ii qr}}{k^2+\ii\epsilon-q^2}
\]
in the complex $q$ plane. For $\epsilon>0$ and $k>0$, its poles lie near
$q=k+\ii\eta$ and $q=-k-\ii\eta$, where
$\eta\simeq\epsilon/(2k)>0$. Since $r>0$, close the contour in the upper
half-plane.
