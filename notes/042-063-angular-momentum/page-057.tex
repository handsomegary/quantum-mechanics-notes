\notepage{57}{Atomic energy, length, and speed scales}
With the same Coulomb-unit convention as the preceding pages,
\begin{align*}
 R_y&=\frac{me^4}{2\hbar^2}
 =\frac12mc^2\left(\frac{e^2}{\hbar c}\right)^2
 =\frac12mc^2\alpha^2,\\
 \alpha&=\frac{e^2}{\hbar c},\qquad
 \alpha^{-1}\simeq137.036.
\end{align*}
\editorial{The source's correction callout explicitly changes the
printed label to $1/\alpha$. The number $137.036$ is the approximate
inverse fine-structure constant, not $\alpha$ itself.}

The rest-energy estimates written in the notes are
\[
 mc^2\simeq0.511\ \mathrm{MeV}\simeq0.5\ \mathrm{MeV}
 \quad\text{(electron)},
\]
\[
 Mc^2\simeq938.3\ \mathrm{MeV}\simeq1000\ \mathrm{MeV}
 \quad\text{(proton)}.
\]
These rounded values illustrate the separation of atomic and
rest-energy scales.

\smallheading{The Bohr length and the decay constant}
\[
 R_y=\frac{e^2}{2a_0},\qquad
 a_0=\frac{\hbar^2}{me^2},\qquad
 \frac1{na_0}=\sqrt{\frac{-2mE_n}{\hbar^2}}.
\]
The Coulomb virial result gives
\[
 \langle T\rangle_{n\ell m}=\frac{R_y}{n^2}.
\]
For the ground state in the nonrelativistic description,
\[
 \frac12m\langle v^2\rangle_{n=1}
 =\frac12mc^2\alpha^2,
 \qquad
 v_{\rm rms}=\sqrt{\langle v^2\rangle_{n=1}}=c\alpha.
\]
Using the conventional dimensionless speed $\beta=v/c$,
\[
 \beta_{\rm rms}=\alpha,\qquad
 R_y=\frac12mc^2\alpha^2
 =\frac12mc^2\beta_{\rm rms}^2.
\]
